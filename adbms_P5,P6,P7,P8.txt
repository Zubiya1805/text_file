# ADBMS PRACTICAL – Q8
## University Campus Locations
### Software: Oracle 10g
NOTE: question 7 (metadata) and 8 (index) are placed before question 5
   because SDO_WITHIN_DISTANCE needs them.

-- 1. Create Campus_Place table
CREATE TABLE CAMPUS_PLACE (
    PLACE_ID NUMBER PRIMARY KEY,
    PLACE_NAME VARCHAR2(50),
    LOCATION MDSYS.SDO_GEOMETRY
);

-- 2. Insert three point locations using SDO_GEOMETRY
INSERT INTO CAMPUS_PLACE VALUES(
    1, 'Administration Building',
    MDSYS.SDO_GEOMETRY(
        2001, 4326,
        MDSYS.SDO_POINT_TYPE(72.8697, 19.0210, NULL),
        NULL, NULL
    )
);

INSERT INTO CAMPUS_PLACE VALUES(
    2, 'Cafeteria',
    MDSYS.SDO_GEOMETRY(
        2001, 4326,
        MDSYS.SDO_POINT_TYPE(72.8700, 19.0213, NULL),
        NULL, NULL
    )
);

INSERT INTO CAMPUS_PLACE VALUES(
    3, 'Sports Complex',
    MDSYS.SDO_GEOMETRY(
        2001, 4326,
        MDSYS.SDO_POINT_TYPE(72.8730, 19.0230, NULL),
        NULL, NULL
    )
);

-- 3. Display the X and Y coordinates of all places
SELECT
    c.PLACE_ID,
    c.PLACE_NAME,
    c.LOCATION.SDO_POINT.X AS X_LONGITUDE,
    c.LOCATION.SDO_POINT.Y AS Y_LATITUDE
FROM CAMPUS_PLACE c;

-- 4. Distance between Administration Building and Cafeteria
SELECT
    a.PLACE_NAME AS PLACE_1,
    b.PLACE_NAME AS PLACE_2,
    ROUND(
        SDO_GEOM.SDO_DISTANCE(
            a.LOCATION,
            b.LOCATION,
            0.0001,
            'unit=M'
        ), 2
    ) AS DISTANCE_METERS
FROM CAMPUS_PLACE a
CROSS JOIN CAMPUS_PLACE b
WHERE a.PLACE_ID = 1
AND b.PLACE_ID = 2;

-- 7. Create and populate USER_SDO_GEOM_METADATA
INSERT INTO USER_SDO_GEOM_METADATA (TABLE_NAME, COLUMN_NAME, DIMINFO, SRID)
VALUES (
    'CAMPUS_PLACE',
    'LOCATION',
    MDSYS.SDO_DIM_ARRAY(
        MDSYS.SDO_DIM_ELEMENT('Longitude', -180, 180, 0.05),
        MDSYS.SDO_DIM_ELEMENT('Latitude', -90, 90, 0.05)
    ),
    4326
);
COMMIT;

-- 8. Create a spatial index on the Location column
CREATE INDEX CAMPUS_PLACE_SIDX
ON CAMPUS_PLACE(LOCATION)
INDEXTYPE IS MDSYS.SPATIAL_INDEX;

-- 5. Places within 300 metres of the Administration Building (SDO_WITHIN_DISTANCE)
SELECT b.PLACE_NAME
FROM CAMPUS_PLACE b
WHERE b.PLACE_NAME <> 'Administration Building'
AND SDO_WITHIN_DISTANCE(
        b.LOCATION,
        (SELECT LOCATION FROM CAMPUS_PLACE WHERE PLACE_NAME = 'Administration Building'),
        'distance=300 unit=M'
    ) = 'TRUE';

-- 6. Distance between Cafeteria and Sports Complex
SELECT
    a.PLACE_NAME AS PLACE_1,
    b.PLACE_NAME AS PLACE_2,
    ROUND(
        SDO_GEOM.SDO_DISTANCE(
            a.LOCATION,
            b.LOCATION,
            0.0001,
            'unit=M'
        ), 2
    ) AS DISTANCE_METERS
FROM CAMPUS_PLACE a
CROSS JOIN CAMPUS_PLACE b
WHERE a.PLACE_ID = 2
AND b.PLACE_ID = 3;

-- 9. Nearest campus place to the Sports Complex
SELECT PLACE_NAME, DISTANCE_METERS
FROM (
    SELECT
        a.PLACE_NAME,
        ROUND(
            SDO_GEOM.SDO_DISTANCE(
                a.LOCATION,
                b.LOCATION,
                0.0001,
                'unit=M'
            ), 2
        ) AS DISTANCE_METERS
    FROM CAMPUS_PLACE a
    CROSS JOIN CAMPUS_PLACE b
    WHERE b.PLACE_ID = 3
    AND a.PLACE_ID <> 3
    ORDER BY DISTANCE_METERS
)
WHERE ROWNUM = 1;

-- 10. Names of places within 1,000 metres of the Cafeteria
SELECT
    a.PLACE_NAME,
    ROUND(
        SDO_GEOM.SDO_DISTANCE(
            a.LOCATION,
            b.LOCATION,
            0.0001,
            'unit=M'
        ), 2
    ) AS DISTANCE_METERS
FROM CAMPUS_PLACE a
CROSS JOIN CAMPUS_PLACE b
WHERE b.PLACE_ID = 2
AND a.PLACE_ID <> 2
AND SDO_GEOM.SDO_DISTANCE(
    a.LOCATION,
    b.LOCATION,
    0.0001,
    'unit=M'
) <= 1000;
